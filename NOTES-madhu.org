** resize a window
#+begin_src lisp
(def-simple-f "window-rules/list-views")

(def-simple-f "window-rules/configure-view"
  "id"
  :&optional
  "geometry"
  "output-id" ;optional-uint64
  "sticky" ;optional-bool
  "tiled-edges" ;optional-uint64
  "fullscreen" ;optional-bool
  )

(defun find-views (c app-id &optional ($ret (window-rules/list-views c)))
  (loop for x across $ret
	if (equal app-id (cdr (assoc "app-id" x :test #'equal)))
	collect x))

(defun view-id (view)
  (cdr (assoc "id" view :test #'equal)))

,#+nil
(window-rules/configure-view
 $c
 'id  (car (mapcar 'view-id (find-views $c "UXTerm")))
 'geometry (geometry-to-json 0 0 800 600))
#+end_src
